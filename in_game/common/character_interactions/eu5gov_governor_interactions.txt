eu5gov_appoint_governor = {
	sound = UI_action_religion_generic
	message = yes
	on_own_nation = yes

	ai_tick = daily
	ai_tick_frequency = 360

	potential = {
		scope:actor = {
			any_owned_location = {
				has_building = building_type:local_governor
				NOT = { has_variable = eu5gov_governor }
			}
		}
	}

	allow = { scope:actor = { always = yes } }

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "eu5gov_select_governor_character"
		none_available_msg_key = "eu5gov_no_governor_character_available"
		column = { data = character_info }
		visible = { eu5gov_can_serve_as_governor = yes }
	}

	select_trigger = {
		looking_for_a = location
		source = actor
		target_flag = target
		name = "eu5gov_select_governor_residence"
		none_available_msg_key = "eu5gov_no_governor_residence_available"
		column = { data = name }
		visible = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			NOT = { has_variable = eu5gov_governor }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:recipient exists = scope:target }
			scope:recipient = { save_scope_as = eu5gov_assignment_governor }
			scope:target = { save_scope_as = eu5gov_assignment_office }
			scope:actor = { eu5gov_assign_governor_to_office_effect = yes }
		}
	}

	ai_will_do = {
		add = scope:recipient.adm
		add = { value = scope:recipient.dip divide = 5 }
		add = { value = scope:recipient.mil divide = 5 }
	}
}

# Cabinet-style replacement path for clicking an occupied portrait.
eu5gov_change_governor_at_residence = {
	sound = UI_action_religion_generic
	message = yes
	on_own_nation = yes
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		exists = scope:target_1
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
		}
	}
	allow = {
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
		}
	}
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "eu5gov_select_replacement_governor"
		none_available_msg_key = "eu5gov_no_governor_character_available"
		column = { data = character_info }
		visible = { eu5gov_can_serve_as_governor = yes }
	}
	effect = {
		if = {
			limit = { exists = scope:recipient exists = scope:target_1 }
			# Replacing a Governor is politically equivalent to dismissing the old one.
			scope:target_1 = {
				var:eu5gov_governor ?= {
					eu5gov_apply_governor_dismissal_cost_effect = yes
				}
				eu5gov_clear_governor_location_effect = yes
				save_scope_as = eu5gov_assignment_office
			}
			scope:recipient = { save_scope_as = eu5gov_assignment_governor }
			scope:actor = { eu5gov_assign_governor_to_office_effect = yes }
		}
	}
	ai_will_do = { add = -100 }
}

eu5gov_dismiss_governor = {
	sound = UI_action_religion_generic
	message = yes
	on_own_nation = yes
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		scope:actor = {
			any_character = {
				is_alive = yes
				has_variable = eu5gov_governorship
			}
		}
	}
	allow = { scope:actor = { always = yes } }
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "eu5gov_select_governor_to_dismiss"
		none_available_msg_key = "eu5gov_no_governor_to_dismiss"
		column = { data = character_info }
		visible = { is_alive = yes has_variable = eu5gov_governorship }
	}
	effect = {
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				save_scope_as = eu5gov_auto_excluded_governor
				eu5gov_apply_governor_dismissal_cost_effect = yes
				var:eu5gov_governorship ?= { eu5gov_clear_governor_location_effect = yes }
			}
			scope:actor = {
				eu5gov_refresh_governor_estate_power_effect = yes
				eu5gov_update_governor_vacancies_effect = yes
			}
		}
	}
	ai_will_do = { add = -100 }
}
