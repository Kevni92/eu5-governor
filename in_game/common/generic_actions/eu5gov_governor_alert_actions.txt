eu5gov_appoint_next_vacant_governor = {
	type = owncountry
	sound = UI_action_religion_generic
	show_message = no
	show_message_to_target = no
	show_in_gui_list = no
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		scope:actor = {
			any_owned_location = {
				has_building = building_type:local_governor
				NOT = { has_variable = eu5gov_governor }
			}
		}
	}

	allow = {
		scope:actor = {
			any_owned_location = {
				has_building = building_type:local_governor
				NOT = { has_variable = eu5gov_governor }
			}
		}
	}

	# Clicking the warning goes straight to the character picker. The target office
	# itself is selected deterministically in the effect below.
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = target
		name = "eu5gov_select_governor_character"
		none_available_msg_key = "eu5gov_no_governor_character_available"
		column = { data = character_info }
		visible = { eu5gov_can_serve_as_governor = yes }
	}

	effect = {
		scope:actor = {
			ordered_owned_location = {
				limit = {
					has_building = building_type:local_governor
					NOT = { has_variable = eu5gov_governor }
				}
				order_by = development
				max = 1
				save_scope_as = eu5gov_next_vacant_governor_office
			}
		}

		if = {
			limit = {
				exists = scope:target
				exists = scope:eu5gov_next_vacant_governor_office
			}

			scope:eu5gov_next_vacant_governor_office = {
				set_variable = { name = eu5gov_governor value = scope:target }
				set_variable = { name = eu5gov_governor_country value = scope:actor }
				set_variable = { name = eu5gov_governor_role value = 1 }
				set_variable = { name = eu5gov_entrenchment value = 0 }
				set_variable = { name = eu5gov_dismissal_cost_scale value = 10 }
				eu5gov_apply_governor_bonus_effect = yes
			}

			scope:target = {
				set_variable = { name = eu5gov_governorship value = scope:eu5gov_next_vacant_governor_office }
				set_variable = { name = eu5gov_governor_role value = 1 }
				set_variable = { name = eu5gov_entrenchment value = 0 }
				set_variable = { name = eu5gov_dismissal_cost_scale value = 10 }
				add_character_modifier = {
					modifier = busy_modifier
					years = -1
					mode = add_and_extend
				}
			}

			scope:actor = {
				remove_from_variable_map = {
					name = eu5gov_governor_roster
					key = scope:target
				}
				add_to_variable_map = {
					name = eu5gov_governor_roster
					key = scope:target
					value = scope:eu5gov_next_vacant_governor_office
				}
				remove_from_variable_map = {
					name = eu5gov_governor_offices
					key = scope:eu5gov_next_vacant_governor_office
				}
				add_to_variable_map = {
					name = eu5gov_governor_offices
					key = scope:eu5gov_next_vacant_governor_office
					value = 1
				}
			}

			scope:eu5gov_next_vacant_governor_office = {
				eu5gov_apply_governor_role_effect = yes
			}
	}

	ai_will_do = { add = -100 }
}
