eu5gov_appoint_governor_from_outliner = {
	type = owncountry
	sound = UI_action_religion_generic
	show_message = no
	show_message_to_target = no
	show_in_gui_list = no
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		exists = scope:target_1
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			NOT = { has_variable = eu5gov_governor }
		}
	}

	allow = {
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			NOT = { has_variable = eu5gov_governor }
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = target
		name = "eu5gov_select_governor_character"
		none_available_msg_key = "eu5gov_no_governor_character_available"
		column = { data = character_info }
		visible = { eu5gov_can_serve_as_governor = yes }
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
				exists = scope:target_1
			}
			scope:target = { save_scope_as = eu5gov_assignment_governor }
			scope:target_1 = { save_scope_as = eu5gov_assignment_office }
			scope:actor = { eu5gov_assign_governor_to_office_effect = yes }
		}
	}

	ai_will_do = { add = -100 }
}


eu5gov_change_governor_from_outliner = {
	type = owncountry
	force_click_and_confirm_or_hold = yes
	sound = UI_action_religion_generic
	show_message = no
	show_message_to_target = no
	show_in_gui_list = no
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		exists = scope:target_1
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
		}
	}

	allow = {
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = target
		name = "eu5gov_select_replacement_governor"
		none_available_msg_key = "eu5gov_no_governor_character_available"
		column = { data = character_info }
		visible = { eu5gov_can_serve_as_governor = yes }
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
				exists = scope:target_1
			}

			scope:target_1 = {
				var:eu5gov_governor ?= {
					if = {
						limit = { has_estate = estate_type:crown_estate }
						custom_tooltip = EU5GOV_REPLACE_CONFIRM_CROWN_FREE
					}
					else = {
						custom_tooltip = {
							text = EU5GOV_REPLACE_CONFIRM_COSTS
							save_scope_as = eu5gov_replaced_governor
							var:eu5gov_governorship ?= {
								owner ?= {
									if = {
										limit = { scope:eu5gov_replaced_governor = { has_variable = eu5gov_dismissal_cost_scale } }
										add_estate_satisfaction = {
											type = scope:eu5gov_replaced_governor.estate_type
											value = { value = scope:eu5gov_replaced_governor.var:eu5gov_dismissal_cost_scale multiply = -0.01 }
										}
										add_stability = { value = scope:eu5gov_replaced_governor.var:eu5gov_dismissal_cost_scale multiply = -0.50 }
									}
									else = {
										add_estate_satisfaction = { type = scope:eu5gov_replaced_governor.estate_type value = -0.10 }
										add_stability = -5
									}
								}
							}
						}
					}
				}
			}

			hidden_effect = {
				scope:target_1 = {
					eu5gov_clear_governor_location_effect = yes
					save_scope_as = eu5gov_assignment_office
				}
				scope:target = { save_scope_as = eu5gov_assignment_governor }
				scope:actor = { eu5gov_assign_governor_to_office_effect = yes }
			}
		}
	}

	ai_will_do = { add = -100 }
}


# Direct outliner actions use the same Generic Action path as the portrait chooser.
# Role changes cost 20 Stability and are disabled when their territorial requirement
# is not met, instead of accepting a click and then silently doing nothing.
eu5gov_set_role_normal_from_outliner = {
	type = owncountry
	force_click_and_confirm_or_hold = yes
	sound = UI_action_religion_generic
	show_message = no
	show_message_to_target = no
	show_in_gui_list = no
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		exists = scope:target_1
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
			NOT = { var:eu5gov_governor_role = 1 }
		}
	}
	allow = { always = yes }
	effect = {
		custom_tooltip = {
			text = EU5GOV_ROLE_CHANGE_CONFIRM_NORMAL
			scope:actor = { add_stability = -20 }
		}
		hidden_effect = {
			scope:target_1 = {
				set_variable = { name = eu5gov_governor_role value = 1 }
				var:eu5gov_governor ?= { set_variable = { name = eu5gov_governor_role value = 1 } }
				eu5gov_apply_governor_role_effect = yes
			}
		}
	}
	ai_will_do = { add = -100 }
}

eu5gov_dismiss_governor_from_outliner = {
	type = owncountry
	force_click_and_confirm_or_hold = yes
	sound = UI_action_religion_generic
	show_message = no
	show_message_to_target = no
	show_in_gui_list = no
	ai_tick = never
	ai_tick_frequency = 99999

	potential = {
		exists = scope:target_1
		scope:target_1 = {
			owner ?= scope:actor
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
		}
	}
	allow = { always = yes }
	effect = {
		scope:target_1 = {
			var:eu5gov_governor ?= {
				save_scope_as = eu5gov_auto_excluded_governor
				if = {
					limit = { has_estate = estate_type:crown_estate }
					custom_tooltip = EU5GOV_DISMISS_CONFIRM_CROWN_FREE
				}
				else = {
					custom_tooltip = {
						text = EU5GOV_DISMISS_CONFIRM_COSTS
						save_scope_as = eu5gov_action_dismissed_governor
						var:eu5gov_governorship ?= {
							owner ?= {
								if = {
									limit = { scope:eu5gov_action_dismissed_governor = { has_variable = eu5gov_dismissal_cost_scale } }
									add_estate_satisfaction = {
										type = scope:eu5gov_action_dismissed_governor.estate_type
										value = { value = scope:eu5gov_action_dismissed_governor.var:eu5gov_dismissal_cost_scale multiply = -0.01 }
									}
									add_stability = { value = scope:eu5gov_action_dismissed_governor.var:eu5gov_dismissal_cost_scale multiply = -0.50 }
								}
								else = {
									add_estate_satisfaction = { type = scope:eu5gov_action_dismissed_governor.estate_type value = -0.10 }
									add_stability = -5
								}
							}
						}
					}
				}
			}
		}
		hidden_effect = {
			scope:target_1 = { eu5gov_clear_governor_location_effect = yes }
			scope:actor = {
				eu5gov_refresh_governor_estate_power_effect = yes
				eu5gov_update_governor_vacancies_effect = yes
			}
		}
	}
	ai_will_do = { add = -100 }
}
