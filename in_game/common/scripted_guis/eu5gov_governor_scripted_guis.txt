eu5gov_set_role_normal = {
	scope = country
	saved_scopes = { office }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		scope:office ?= {
			if = {
				limit = {
					has_variable = eu5gov_governor
					NOT = { var:eu5gov_governor_role = 1 }
				}
				set_variable = { name = eu5gov_governor_role value = 1 }
				var:eu5gov_governor ?= {
					set_variable = { name = eu5gov_governor_role value = 1 }
				}
				eu5gov_apply_governor_role_effect = yes
			}
		}
	}
}

eu5gov_dismiss_governor_sgui = {
	scope = country
	saved_scopes = { office }
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		scope:office ?= {
			if = {
				limit = { has_variable = eu5gov_governor }
				var:eu5gov_governor ?= {
					eu5gov_apply_governor_dismissal_cost_effect = yes
				}
				eu5gov_clear_governor_location_effect = yes
				owner ?= {
					eu5gov_refresh_governor_estate_power_effect = yes
				}
			}
		}
	}
}
