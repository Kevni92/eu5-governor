eu5gov_set_role_normal = {
	scope = country
	saved_scopes = { governor }
	is_shown = {
		scope:governor ?= {
			is_alive = yes
			has_variable = eu5gov_governorship
		}
	}
	is_valid = {
		scope:governor ?= {
			is_alive = yes
			has_variable = eu5gov_governorship
		}
	}
	effect = {
		scope:governor = {
			set_variable = { name = eu5gov_governor_role value = 1 }
			var:eu5gov_governorship ?= {
				set_variable = { name = eu5gov_governor_role value = 1 }
				eu5gov_apply_governor_role_effect = yes
			}
		}
	}
}
