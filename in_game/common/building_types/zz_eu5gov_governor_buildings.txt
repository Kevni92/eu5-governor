# Character Governors
# Keep the vanilla local_governor id so existing unlocks, scripted references and AI logic
# continue to point at the unified governor building.

REPLACE:local_governor = {
	audio_tier = 2
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes

	employment_size = large_noble_employment
	category = government_category

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	allow = {
		custom_tooltip = {
			text = eu5gov_governor_capacity_tt
			scope:actor = {
				total_building_levels_including_construction:local_governor < {
					value = modifier:num_local_governors
					add = modifier:num_naval_governors
					subtract = total_building_levels_including_construction:lieutenancy
				}
			}
		}
		owner = scope:actor
		trigger_if = {
			limit = { has_building = building_type:seljuk_mint }
			NOT = { has_building = building_type:seljuk_mint }
		}
	}

	location_potential = {
		owner.capital != this
	}

	expensive = yes
	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes
	important_for_AI = yes
	ai_ignore_maintenance = yes
	build_time = government_build_time

	unique_production_methods = {
		local_governor_maintenance = {
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 3
			books = 1
			goods_gold = 1
			furniture = 2
			fine_cloth = 1
			category = building_maintenance
		}
	}

	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 30
	}

	# Make a newly completed residence appear immediately as a vacant Governor slot.
	on_built = {
		location = { save_scope_as = eu5gov_built_office_location }
		owner = {
			remove_from_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_built_office_location
			}
			add_to_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_built_office_location
				value = 1
			}
			eu5gov_update_governor_vacancies_effect = yes
		}
	}

	# Demolishing the residence removes the slot and vacates any serving Governor.
	# This is not a voluntary dismissal, so no political dismissal cost is charged.
	on_destroyed = {
		location = {
			save_scope_as = eu5gov_destroyed_office_location
			if = {
				limit = { has_variable = eu5gov_governor }
				eu5gov_clear_governor_location_effect = yes
			}
		}
		owner = {
			remove_from_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_destroyed_office_location
			}
			eu5gov_refresh_governor_estate_power_effect = yes
			eu5gov_update_governor_vacancies_effect = yes
		}
	}
}

# Maritime and land governorship are one institution in this mod. Existing naval governor
# capacity is consumed by local_governor above; the old building itself is retired.
REPLACE:naval_governor = {
	audio_tier = 2
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = no

	employment_size = large_noble_employment
	category = government_category

	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes

	allow = { always = no }
	country_potential = { always = no }
	location_potential = { always = no }
	remove_if = { always = yes }

	expensive = yes
	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes
	build_time = government_build_time

	unique_production_methods = {
		naval_governor_maintenance = {
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 3
			books = 1
			goods_gold = 1
			furniture = 2
			naval_supplies = 0.5
			fine_cloth = 0.5
			category = building_maintenance
		}
	}

	construction_demand = local_governor_construction
	modifier = { local_proximity_source = 30 }
}
