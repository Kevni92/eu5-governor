# Parent on-action bridges.
on_character_death = {
	on_actions = {
		eu5gov_governor_death
	}
}

monthly_country_pulse = {
	on_actions = {
		eu5gov_governor_monthly_entrenchment
	}
}

yearly_country_pulse = {
	on_actions = {
		eu5gov_governor_yearly_refresh
	}
}

# root = country, scope:target = character who died
eu5gov_governor_death = {
	trigger = {
		scope:target = { has_variable = eu5gov_governorship }
	}
	effect = {
		scope:target = {
			var:eu5gov_governorship ?= {
				eu5gov_clear_governor_location_effect = yes
			}
		}
		# Death is not a voluntary dismissal, but its Estate power contribution
		# must disappear immediately.
		eu5gov_refresh_governor_estate_power_effect = yes
	}
}


# root = country. Advance serving Governors once per monthly country pulse using
# (2 + (ADM + DIP + MIL) / 100) / 12, preserving the previous annual balance.
eu5gov_governor_monthly_entrenchment = {
	effect = {
		save_scope_as = eu5gov_monthly_country
		every_owned_location = {
			limit = {
				has_building = building_type:local_governor
				has_variable = eu5gov_governor
				var:eu5gov_governor ?= { is_alive = yes }
				var:eu5gov_governor_country ?= { this = scope:eu5gov_monthly_country }
			}
			var:eu5gov_governor ?= {
				if = {
					limit = { NOT = { has_variable = eu5gov_entrenchment } }
					set_variable = { name = eu5gov_entrenchment value = 0 }
				}
				eu5gov_tick_governor_entrenchment_effect = yes
			}
		}
		eu5gov_refresh_governor_estate_power_effect = yes
	}
}

# Rebuild the complete office roster, refresh live bonuses, repair Entrenchment state,
# recalculate Estate power and clean stale assignments. Existing saves are migrated
# lazily from owned buildings.
eu5gov_governor_yearly_refresh = {
	effect = {
		save_scope_as = eu5gov_refresh_country
		eu5gov_rebuild_governor_office_roster_effect = yes

		every_owned_location = {
			limit = { has_variable = eu5gov_governor }
			if = {
				limit = {
					has_building = building_type:local_governor
					var:eu5gov_governor ?= { is_alive = yes }
					var:eu5gov_governor_country ?= { this = scope:eu5gov_refresh_country }
				}

				var:eu5gov_governor ?= {
					if = {
						limit = { NOT = { has_variable = eu5gov_governor_role } }
						set_variable = { name = eu5gov_governor_role value = 1 }
					}
					if = {
						limit = { NOT = { has_variable = eu5gov_entrenchment } }
						set_variable = { name = eu5gov_entrenchment value = 0 }
					}
				}

				eu5gov_register_governor_in_roster_effect = yes
				eu5gov_apply_governor_bonus_effect = yes
				eu5gov_apply_governor_role_effect = yes
			}
			else = {
				eu5gov_clear_governor_location_effect = yes
			}
		}

		eu5gov_refresh_governor_estate_power_effect = yes
	}
}
