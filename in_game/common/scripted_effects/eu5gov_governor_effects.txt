# Current scope: location with var:eu5gov_governor pointing to the appointed character.
eu5gov_apply_governor_bonus_effect = {
	if = {
		limit = { has_location_modifier = eu5gov_governor_administration }
		remove_location_modifier = eu5gov_governor_administration
	}

	if = {
		limit = {
			has_variable = eu5gov_governor
			var:eu5gov_governor ?= { is_alive = yes }
		}
		add_location_modifier = {
			modifier = eu5gov_governor_administration
			mode = add_and_extend
			size = {
				value = var:eu5gov_governor.adm
				multiply = 0.25
				add = {
					value = var:eu5gov_governor.dip
					multiply = 0.05
				}
				add = {
					value = var:eu5gov_governor.mil
					multiply = 0.05
				}
			}
		}
	}
}

# Current scope: governor location. Role 1 = Normal, 2 = Integration, 3 = Colonial.
eu5gov_apply_governor_role_effect = {
	if = {
		limit = { has_location_modifier = eu5gov_integration_governor }
		remove_location_modifier = eu5gov_integration_governor
	}
	if = {
		limit = { has_location_modifier = eu5gov_colonial_governor }
		remove_location_modifier = eu5gov_colonial_governor
	}

	if = {
		limit = {
			has_variable = eu5gov_governor
			NOT = { dominant_culture = owner.culture }
			var:eu5gov_governor ?= {
				is_alive = yes
				var:eu5gov_governor_role = 2
			}
		}
		add_location_modifier = {
			modifier = eu5gov_integration_governor
			mode = add_and_extend
		}
	}
	else_if = {
		limit = {
			has_variable = eu5gov_governor
			is_overseas_for_owner = yes
			var:eu5gov_governor ?= {
				is_alive = yes
				var:eu5gov_governor_role = 3
			}
		}
		add_location_modifier = {
			modifier = eu5gov_colonial_governor
			mode = add_and_extend
		}
	}
}

# Current scope: serving governor character. Time alone entrenches an office holder;
# high total ADM+DIP+MIL accelerates the yearly gain. The same value is mirrored
# onto the governed location so the office and character both carry the tenure state.
eu5gov_tick_governor_entrenchment_effect = {
	if = {
		limit = { NOT = { has_variable = eu5gov_entrenchment } }
		set_variable = { name = eu5gov_entrenchment value = 0 }
	}

	change_variable = { name = eu5gov_entrenchment add = 2 }
	change_variable = {
		name = eu5gov_entrenchment
		add = {
			value = adm
			divide = 100
		}
	}
	change_variable = {
		name = eu5gov_entrenchment
		add = {
			value = dip
			divide = 100
		}
	}
	change_variable = {
		name = eu5gov_entrenchment
		add = {
			value = mil
			divide = 100
		}
	}
	clamp_variable = { name = eu5gov_entrenchment min = 0 max = 100 }

	save_scope_as = eu5gov_entrenchment_governor
	var:eu5gov_governorship ?= {
		set_variable = {
			name = eu5gov_entrenchment
			value = scope:eu5gov_entrenchment_governor.var:eu5gov_entrenchment
		}
		set_variable = {
			name = eu5gov_governor_role
			value = scope:eu5gov_entrenchment_governor.var:eu5gov_governor_role
		}
	}
}

# Current scope: country. The office map is authoritative for the UI and contains
# every owned Governor's Residence, including residences without an appointed character.
eu5gov_rebuild_governor_office_roster_effect = {
	clear_variable_map = eu5gov_governor_offices
	save_scope_as = eu5gov_office_country
	every_owned_location = {
		limit = { has_building = building_type:local_governor }
		save_scope_as = eu5gov_office_location
		scope:eu5gov_office_country = {
			add_to_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_office_location
				value = 1
			}
		}
	}
}

# Current scope: governor location. Keeps the country-side active-Governor roster
# and the all-office roster used by the UI in sync.
eu5gov_register_governor_in_roster_effect = {
	if = {
		limit = {
			has_variable = eu5gov_governor
			has_variable = eu5gov_governor_country
		}
		save_scope_as = eu5gov_roster_location
		var:eu5gov_governor ?= { save_scope_as = eu5gov_roster_character }
		var:eu5gov_governor_country ?= {
			remove_from_variable_map = {
				name = eu5gov_governor_roster
				key = scope:eu5gov_roster_character
			}
			add_to_variable_map = {
				name = eu5gov_governor_roster
				key = scope:eu5gov_roster_character
				value = scope:eu5gov_roster_location
			}
			remove_from_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_roster_location
			}
			add_to_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_roster_location
				value = 1
			}
		}
	}
}

eu5gov_clear_governor_location_effect = {
	if = {
		limit = { has_variable = eu5gov_governor }
		var:eu5gov_governor ?= { save_scope_as = eu5gov_clear_governor_character }

		if = {
			limit = { has_variable = eu5gov_governor_country }
			var:eu5gov_governor_country ?= {
				remove_from_variable_map = {
					name = eu5gov_governor_roster
					key = scope:eu5gov_clear_governor_character
				}
			}
		}

		scope:eu5gov_clear_governor_character = {
			remove_character_modifier = busy_modifier
			remove_variable = eu5gov_governorship
			remove_variable = eu5gov_governor_role
			remove_variable = eu5gov_entrenchment
		}
	}

	if = {
		limit = { has_location_modifier = eu5gov_governor_administration }
		remove_location_modifier = eu5gov_governor_administration
	}
	if = {
		limit = { has_location_modifier = eu5gov_integration_governor }
		remove_location_modifier = eu5gov_integration_governor
	}
	if = {
		limit = { has_location_modifier = eu5gov_colonial_governor }
		remove_location_modifier = eu5gov_colonial_governor
	}

	remove_variable = eu5gov_governor
	remove_variable = eu5gov_governor_country
	remove_variable = eu5gov_governor_role
	remove_variable = eu5gov_entrenchment
}
