# Current scope: location with var:eu5gov_governor pointing to the appointed character.
eu5gov_apply_governor_bonus_effect = {
	if = {
		limit = { has_location_modifier = eu5gov_governor_administration }
		remove_location_modifier = eu5gov_governor_administration
	}

	if = {
		limit = {
			has_variable = eu5gov_governor
			var:eu5gov_governor ?= { is_alive = yes }
		}
		add_location_modifier = {
			modifier = eu5gov_governor_administration
			mode = add_and_extend
			size = {
				value = var:eu5gov_governor.adm
				multiply = 0.25
				add = {
					value = var:eu5gov_governor.dip
					multiply = 0.05
				}
				add = {
					value = var:eu5gov_governor.mil
					multiply = 0.05
				}
			}
		}
	}
}

# Current scope: governor location. Role 1 = Normal, 2 = Integration, 3 = Colonial.
eu5gov_apply_governor_role_effect = {
	if = {
		limit = { has_location_modifier = eu5gov_integration_governor }
		remove_location_modifier = eu5gov_integration_governor
	}
	if = {
		limit = { has_location_modifier = eu5gov_colonial_governor }
		remove_location_modifier = eu5gov_colonial_governor
	}

	if = {
		limit = {
			has_variable = eu5gov_governor
			NOT = { dominant_culture = owner.culture }
			var:eu5gov_governor ?= {
				is_alive = yes
				var:eu5gov_governor_role = 2
			}
		}
		add_location_modifier = {
			modifier = eu5gov_integration_governor
			mode = add_and_extend
		}
	}
	else_if = {
		limit = {
			has_variable = eu5gov_governor
			is_overseas_for_owner = yes
			var:eu5gov_governor ?= {
				is_alive = yes
				var:eu5gov_governor_role = 3
			}
		}
		add_location_modifier = {
			modifier = eu5gov_colonial_governor
			mode = add_and_extend
		}
	}
}

# Current scope: serving governor character. Time alone entrenches an office holder;
# high total ADM+DIP+MIL accelerates the yearly gain. The same value is mirrored
# onto the governed location so the office and character both carry the tenure state.
eu5gov_tick_governor_entrenchment_effect = {
	if = {
		limit = { NOT = { has_variable = eu5gov_entrenchment } }
		set_variable = { name = eu5gov_entrenchment value = 0 }
	}

	change_variable = { name = eu5gov_entrenchment add = 2 }
	change_variable = {
		name = eu5gov_entrenchment
		add = {
			value = adm
			divide = 100
		}
	}
	change_variable = {
		name = eu5gov_entrenchment
		add = {
			value = dip
			divide = 100
		}
	}
	change_variable = {
		name = eu5gov_entrenchment
		add = {
			value = mil
			divide = 100
		}
	}
	clamp_variable = { name = eu5gov_entrenchment min = 0 max = 100 }

	save_scope_as = eu5gov_entrenchment_governor
	var:eu5gov_governorship ?= {
		set_variable = {
			name = eu5gov_entrenchment
			value = scope:eu5gov_entrenchment_governor.var:eu5gov_entrenchment
		}
		set_variable = {
			name = eu5gov_governor_role
			value = scope:eu5gov_entrenchment_governor.var:eu5gov_governor_role
		}
	}
}

# Current scope: serving governor character.
# Voluntary dismissal/replacement is free for Crown characters. Otherwise the
# Governor's Estate loses up to 25 percentage points of satisfaction and the
# country loses up to 10 Legitimacy, linearly scaled by Entrenchment/Machtbasis.
eu5gov_apply_governor_dismissal_cost_effect = {
	if = {
		limit = {
			has_variable = eu5gov_governorship
			has_variable = eu5gov_entrenchment
			NOT = { has_estate = estate_type:crown_estate }
		}
		save_scope_as = eu5gov_dismissed_governor
		var:eu5gov_governorship ?= {
			owner ?= {
				add_estate_satisfaction = {
					type = scope:eu5gov_dismissed_governor.estate_type
					value = {
						value = scope:eu5gov_dismissed_governor.var:eu5gov_entrenchment
						multiply = -0.0025
					}
				}
				add_legitimacy = {
					value = scope:eu5gov_dismissed_governor.var:eu5gov_entrenchment
					multiply = -0.10
				}
			}
		}
	}
}

# Current scope: country.
# Each serving Governor increases the power of their Estate by half their
# Entrenchment percentage. Example: 100 Entrenchment => +50% Estate power.
# Governors of the same Estate stack additively into one deterministic modifier.
eu5gov_refresh_governor_estate_power_effect = {
	if = { limit = { has_country_modifier = eu5gov_governor_crown_estate_power } remove_country_modifier = eu5gov_governor_crown_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_nobles_estate_power } remove_country_modifier = eu5gov_governor_nobles_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_clergy_estate_power } remove_country_modifier = eu5gov_governor_clergy_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_burghers_estate_power } remove_country_modifier = eu5gov_governor_burghers_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_peasants_estate_power } remove_country_modifier = eu5gov_governor_peasants_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_tribes_estate_power } remove_country_modifier = eu5gov_governor_tribes_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_cossacks_estate_power } remove_country_modifier = eu5gov_governor_cossacks_estate_power }
	if = { limit = { has_country_modifier = eu5gov_governor_dhimmi_estate_power } remove_country_modifier = eu5gov_governor_dhimmi_estate_power }

	set_variable = { name = eu5gov_power_crown value = 0 }
	set_variable = { name = eu5gov_power_nobles value = 0 }
	set_variable = { name = eu5gov_power_clergy value = 0 }
	set_variable = { name = eu5gov_power_burghers value = 0 }
	set_variable = { name = eu5gov_power_peasants value = 0 }
	set_variable = { name = eu5gov_power_tribes value = 0 }
	set_variable = { name = eu5gov_power_cossacks value = 0 }
	set_variable = { name = eu5gov_power_dhimmi value = 0 }

	save_scope_as = eu5gov_estate_power_country
	every_owned_location = {
		limit = {
			has_building = building_type:local_governor
			has_variable = eu5gov_governor
			var:eu5gov_governor ?= {
				is_alive = yes
				has_variable = eu5gov_entrenchment
			}
		}
		var:eu5gov_governor ?= { save_scope_as = eu5gov_estate_power_governor }
		scope:eu5gov_estate_power_country = {
			if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:crown_estate } }
				change_variable = { name = eu5gov_power_crown add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:nobles_estate } }
				change_variable = { name = eu5gov_power_nobles add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:clergy_estate } }
				change_variable = { name = eu5gov_power_clergy add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:burghers_estate } }
				change_variable = { name = eu5gov_power_burghers add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:peasants_estate } }
				change_variable = { name = eu5gov_power_peasants add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:tribes_estate } }
				change_variable = { name = eu5gov_power_tribes add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:cossacks_estate } }
				change_variable = { name = eu5gov_power_cossacks add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
			else_if = {
				limit = { scope:eu5gov_estate_power_governor = { has_estate = estate_type:dhimmi_estate } }
				change_variable = { name = eu5gov_power_dhimmi add = { value = scope:eu5gov_estate_power_governor.var:eu5gov_entrenchment divide = 200 } }
			}
		}
	}

	if = {
		limit = { var:eu5gov_power_crown > 0 }
		add_country_modifier = { modifier = eu5gov_governor_crown_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_crown } }
	}
	if = {
		limit = { var:eu5gov_power_nobles > 0 }
		add_country_modifier = { modifier = eu5gov_governor_nobles_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_nobles } }
	}
	if = {
		limit = { var:eu5gov_power_clergy > 0 }
		add_country_modifier = { modifier = eu5gov_governor_clergy_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_clergy } }
	}
	if = {
		limit = { var:eu5gov_power_burghers > 0 }
		add_country_modifier = { modifier = eu5gov_governor_burghers_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_burghers } }
	}
	if = {
		limit = { var:eu5gov_power_peasants > 0 }
		add_country_modifier = { modifier = eu5gov_governor_peasants_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_peasants } }
	}
	if = {
		limit = { var:eu5gov_power_tribes > 0 }
		add_country_modifier = { modifier = eu5gov_governor_tribes_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_tribes } }
	}
	if = {
		limit = { var:eu5gov_power_cossacks > 0 }
		add_country_modifier = { modifier = eu5gov_governor_cossacks_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_cossacks } }
	}
	if = {
		limit = { var:eu5gov_power_dhimmi > 0 }
		add_country_modifier = { modifier = eu5gov_governor_dhimmi_estate_power years = -1 mode = add_and_extend size = { value = var:eu5gov_power_dhimmi } }
	}

	remove_variable = eu5gov_power_crown
	remove_variable = eu5gov_power_nobles
	remove_variable = eu5gov_power_clergy
	remove_variable = eu5gov_power_burghers
	remove_variable = eu5gov_power_peasants
	remove_variable = eu5gov_power_tribes
	remove_variable = eu5gov_power_cossacks
	remove_variable = eu5gov_power_dhimmi
}

# Current scope: country. The office map is authoritative for the UI and contains
# every owned Governor's Residence, including residences without an appointed character.
eu5gov_rebuild_governor_office_roster_effect = {
	clear_variable_map = eu5gov_governor_offices
	save_scope_as = eu5gov_office_country
	every_owned_location = {
		limit = { has_building = building_type:local_governor }
		save_scope_as = eu5gov_office_location
		scope:eu5gov_office_country = {
			add_to_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_office_location
				value = 1
			}
		}
	}
}

# Current scope: governor location. Keeps the country-side active-Governor roster
# and the all-office roster used by the UI in sync.
eu5gov_register_governor_in_roster_effect = {
	if = {
		limit = {
			has_variable = eu5gov_governor
			has_variable = eu5gov_governor_country
		}
		save_scope_as = eu5gov_roster_location
		var:eu5gov_governor ?= { save_scope_as = eu5gov_roster_character }
		var:eu5gov_governor_country ?= {
			remove_from_variable_map = {
				name = eu5gov_governor_roster
				key = scope:eu5gov_roster_character
			}
			add_to_variable_map = {
				name = eu5gov_governor_roster
				key = scope:eu5gov_roster_character
				value = scope:eu5gov_roster_location
			}
			remove_from_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_roster_location
			}
			add_to_variable_map = {
				name = eu5gov_governor_offices
				key = scope:eu5gov_roster_location
				value = 1
			}
		}
	}
}

eu5gov_clear_governor_location_effect = {
	if = {
		limit = { has_variable = eu5gov_governor }
		var:eu5gov_governor ?= { save_scope_as = eu5gov_clear_governor_character }

		if = {
			limit = { has_variable = eu5gov_governor_country }
			var:eu5gov_governor_country ?= {
				remove_from_variable_map = {
					name = eu5gov_governor_roster
					key = scope:eu5gov_clear_governor_character
				}
			}
		}

		scope:eu5gov_clear_governor_character = {
			remove_character_modifier = busy_modifier
			remove_variable = eu5gov_governorship
			remove_variable = eu5gov_governor_role
			remove_variable = eu5gov_entrenchment
		}
	}

	if = {
		limit = { has_location_modifier = eu5gov_governor_administration }
		remove_location_modifier = eu5gov_governor_administration
	}
	if = {
		limit = { has_location_modifier = eu5gov_integration_governor }
		remove_location_modifier = eu5gov_integration_governor
	}
	if = {
		limit = { has_location_modifier = eu5gov_colonial_governor }
		remove_location_modifier = eu5gov_colonial_governor
	}

	remove_variable = eu5gov_governor
	remove_variable = eu5gov_governor_country
	remove_variable = eu5gov_governor_role
	remove_variable = eu5gov_entrenchment
}
